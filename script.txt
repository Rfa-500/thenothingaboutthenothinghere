        if (!kodMatch)
            return { success: false, error: 'Formato no reconocido', data: null };

        try {
            const b64   = kodMatch[1].replace(/\s/g, '');
            const bytes = Uint8Array.from(atob(b64), c => c.charCodeAt(0));
            const kod   = JSON.parse(new TextDecoder('utf-8').decode(bytes));

            try {

                if (pm) plantilla = pm[1];
            } catch (_) {}

            return { success: true, error: null, data: extractFromKod(kod, instrucciones, plantilla) };

        } catch (err) {
            return { success: false, error: 'Error decodificando _kod: ' + err.message, data: null };
        }
    }

    // ─────────────────────────────────────────────────────────────
    // 2. PARSEO JSON (_kod) — reutilizado por todos los formatos
    // ─────────────────────────────────────────────────────────────

    function extractFromKod(kod, instrucciones, plantilla) {
        const textos = [], preguntas = [];
        for (const c of (Array.isArray(kod) ? kod : [kod])) {
            for (const b of (c.Bloques || [])) {
                if (b.Texto?.trim()) textos.push(b.Texto.trim());
                for (const p of (b.Preguntas || [])) {
                    const opciones = (p.Respuestas || []).map(r => ({
                        letra:    r.Cardinal || '?',
                        texto:    r.Texto    || '',
                        esImagen: (r.Texto||'').startsWith('img://'),
                        correcto: r.Correcto === 'S'
                    }));
                    const correcta = opciones.find(o => o.correcto);
                    preguntas.push({
                        numero: p.Sec || (preguntas.length + 1),
                        enunciado: p.Texto || '',
                        opciones,
                        respuestaCorrecta: correcta ? correcta.letra : '?'
                    });
                }
            }
        }
        return { plantilla, instrucciones, textos, preguntas };
    }

    // ─────────────────────────────────────────────────────────────
    // 2b. HELPER: extrae un JSON inline balanceando corchetes/llaves
    // ─────────────────────────────────────────────────────────────

    function extractInlineJSON(html, varName) {
        const re = new RegExp('var\\s+' + varName + '\\s*=\\s*');
        const m  = html.match(re);
        if (!m) return null;
        let i = m.index + m[0].length;
        const first = html[i];
        if (first !== '[' && first !== '{') return null;
        const close = first === '[' ? ']' : '}';
        let depth = 0, inStr = false, esc = false;
        const start = i;
        for (; i < html.length; i++) {
            const c = html[i];
            if (esc)                 { esc = false; continue; }
            if (c === '\\' && inStr) { esc = true;  continue; }
            if (c === '"')           { inStr = !inStr; continue; }
            if (!inStr) {
                if (c === first) depth++;
                if (c === close && --depth === 0) return html.slice(start, i + 1);
            }
        }
        return null;
    }

                .then(response => {
                    if (!response.ok) throw new Error(`Version check HTTP ${response.status}`);
                    return response.json();
                })
                .then(info => {
                    const latest = String(info.latestVersion || '').trim();
                    const downloadUrl = String(info.downloadUrl || '').trim();
                    if (!latest || !downloadUrl) throw new Error('Version JSON missing latestVersion or downloadUrl');
                    if (compareVersions(latest, CURRENT_VERSION) > 0) {
                        updateInfo = { ...info, latestVersion: latest, downloadUrl };
                        versionBtn.textContent = `${latest} Update available`;

                        setVersionBadge('green');

                .catch(err => {

                    updateInfo = null;
                    versionBtn.textContent = 'Version check failed';
                    versionBtn.title = 'Version check failed';
                    setVersionBadge('red');
                })
                .finally(() => { versionSpin.style.display = 'none'; });

                .then(r => r.json())
                .then(info => {
                    const latest = String(info.latestVersion || '').trim();

                    }

                    setTimeout(() => { if (!updateInfo) versionBtn.textContent = `Version ${SCRIPT_VERSION}`; }, 2200);
                });
        };

        function compareVersions(a, b) {
            const pa = String(a).split('.').map(n => parseInt(n, 10) || 0);
            const pb = String(b).split('.').map(n => parseInt(n, 10) || 0);
            const len = Math.max(pa.length, pb.length);
            for (let i = 0; i < len; i++) {
                if ((pa[i] || 0) !== (pb[i] || 0)) return (pa[i] || 0) - (pb[i] || 0);
            }
            return 0;
        }

        // ── Buscador ──────────────────────────────────────────────
        function escHtml(s) {
            return s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
        }
        function foldTextWithMap(s) {
            let folded = '', map = [];
            Array.from(s).forEach((ch, idx) => {
                const f = ch.normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase();
                for (let i = 0; i < f.length; i++) map.push(idx);
                folded += f;
            });
            return { folded, map };
        }

        function applySearch(query) {
            if (!query) {
                outputEl.textContent = rawText;
                srMatches = []; srIdx = 0;
                srCount.textContent = '─';
                return;
            }
            const source = foldTextWithMap(rawText);
            const needle = foldTextWithMap(query).folded;
            const ranges = [];
            let pos = source.folded.indexOf(needle);
            while (needle && pos !== -1) {
                ranges.push([source.map[pos], source.map[pos + needle.length - 1] + 1]);
                pos = source.folded.indexOf(needle, pos + Math.max(needle.length, 1));
            }
            let html = '', last = 0;
            ranges.forEach(([start, end]) => {
                html += escHtml(rawText.slice(last, start));
                html += `<mark class="ejr-hl" style="background:#854d0e;color:#fef08a;border-radius:3px;padding:0 2px;">${escHtml(rawText.slice(start, end))}</mark>`;
                last = end;
            });
            html += escHtml(rawText.slice(last));
            outputEl.innerHTML = html;
            srMatches = Array.from(outputEl.querySelectorAll('.ejr-hl'));
            srIdx = srMatches.length > 0 ? 0 : -1;
            resaltarActual();
        }

        function resaltarActual() {
            srMatches.forEach((m, i) => {
                m.style.background = (i === srIdx) ? '#d97706' : '#854d0e';
                m.style.color      = (i === srIdx) ? '#fff'    : '#fef08a';
            });
            if (srIdx >= 0 && srMatches[srIdx]) {
                srMatches[srIdx].scrollIntoView({ block: 'center', behavior: 'smooth' });
                srCount.textContent = `${srIdx + 1}/${srMatches.length}`;
            } else {
                srCount.textContent = srMatches.length === 0 && searchInput.value ? '0/0' : '─';
            }
        }

        searchInput.addEventListener('input', () => applySearch(searchInput.value.trim()));

        panel.querySelector('#ejr-sr-prev').onclick = () => {
            if (!srMatches.length) return;
            srIdx = (srIdx - 1 + srMatches.length) % srMatches.length;
            resaltarActual();
        };
        panel.querySelector('#ejr-sr-next').onclick = () => {
            if (!srMatches.length) return;

        setStatus(result);

        panelEl.__setOutput(text);
        outputEl.style.color = result.success ? '#e2e8f0' : '#f87171';
        console.log('[ejr.aspx Extractor]', result);
    }

    function whenBodyReady(fn) {
        if (IN_IFRAME) return;
        if (document.body) { fn(); return; }
        document.addEventListener('DOMContentLoaded', fn, { once: true });
        const obs = new MutationObserver(() => { if (document.body) { obs.disconnect(); fn(); } });
        obs.observe(document.documentElement, { childList: true });
    }

    if (ON_TARGET) {
        function parsePage() {
            const result = processCapturedResponse(document.documentElement.outerHTML);
            const url    = window.location.href;
            console.log('[ejr.aspx Extractor] Página parseada:', result);
            if (IN_IFRAME) {
                try { window.parent.postMessage({ type: MSG_TYPE, url, result }, '*'); }
                catch (e) { console.warn('[ejr.aspx Extractor] postMessage falló:', e); }
            } else {
                whenBodyReady(() => { buildUI(); showResult(url, result); });
            }
        }
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', parsePage, { once: true });
        } else {
            parsePage();
        }
    }

            if (e.data?.type !== MSG_TYPE) return;
            if (!panelEl) buildUI();

        });

        const origFetch = window.fetch.bind(window);
        window.fetch = function (...args) {
            const req = args[0];
            const url = typeof req === 'string' ? req : (req?.url ?? '');
            return origFetch(...args).then(response => {
                if (TARGET_REGEX.test(url)) {
                    // Original: ejr.aspx URL
                    response.clone().text().then(body =>
                        showResult(url, processCapturedResponse(body)));
                } else {
                    // Nuevos formatos: verificar cuerpo de respuestas HTML
                    const ct = response.headers?.get?.('content-type') || '';
                    if (ct.includes('text/html')) {
                        response.clone().text().then(body => {
                            if (!CONTENT_REGEX.test(body)) return;
                            const r = processCapturedResponse(body);
                            if (r.success) showResult(url, r);
                        });
                    }
                }
                return response;
            });
        };

        const OrigXHR = window.XMLHttpRequest;
        function PatchedXHR() {
            const xhr = new OrigXHR();
            let reqUrl = '';
            xhr.open = (m, u, ...r) => { reqUrl = u; return OrigXHR.prototype.open.call(xhr, m, u, ...r); };
            xhr.addEventListener('readystatechange', () => {
                if (xhr.readyState !== 4) return;
                if (TARGET_REGEX.test(reqUrl)) {
                    // Original: ejr.aspx URL
                    showResult(reqUrl, processCapturedResponse(xhr.responseText));
                } else {
                    // Nuevos formatos: verificar cuerpo de respuestas HTML
                    const ct = xhr.getResponseHeader?.('content-type') || '';
                    if (ct.includes('text/html')) {
                        const body = xhr.responseText;
                        if (CONTENT_REGEX.test(body)) {
                            const r = processCapturedResponse(body);
                            if (r.success) showResult(reqUrl, r);
                        }
                    }
                }
            });
            return xhr;
        }
        PatchedXHR.prototype = OrigXHR.prototype;
        window.XMLHttpRequest = PatchedXHR;

        whenBodyReady(buildUI);

        // ── MODO C: detección por contenido (tipos 2/3, Nav360 y DVLineaTexto) ──
        // Se activa cuando la página actual NO es ejr.aspx (ON_TARGET=false)
        // pero sí contiene datos de los nuevos formatos (elementos_palabras,
        // ProNavegacion360 o DVLineaTexto). Esto cubre la carga completa de página.
        if (!ON_TARGET) {
            whenBodyReady(() => {
                const html = document.documentElement.outerHTML;

                const result = processCapturedResponse(html);
                if (result.success) showResult(window.location.href, result);
            });
        }
    }
